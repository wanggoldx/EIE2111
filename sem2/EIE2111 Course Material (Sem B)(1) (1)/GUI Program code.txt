using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

using System.IO;

namespace ReadWriteTextFiles
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            String line;
            try
            {
                StreamReader sr = new StreamReader(textBox1.Text);
                StreamWriter sw = new StreamWriter(textBox2.Text);

                line = sr.ReadLine();
                while (line != null)
                {
                    MessageBox.Show(line, "File Content", MessageBoxButtons.OK);
                    sw.WriteLine(line);
                    line = sr.ReadLine();
                }
                  
                sr.Close();
                sw.Close();
            }
            finally
            {
                MessageBox.Show("Last one", "End", MessageBoxButtons.OK);
            }
        }
    }
}

//Form2.cs (for the constructor and the function button1_Click only)
        public Form2(int c)
        {
            InitializeComponent();
            card = c;
            face = 0;
            ShowCard();
        }

        private int card, face;



        void ShowCard()
        {
            if (card == 0) pictureBox1.Visible = false;
            if (card == 1)
            {
                pictureBox1.Visible = true;
                pictureBox1.Image = Image.FromFile("images\\s1.jpg");
            }
            if (card == 2)
            {
                pictureBox1.Visible = true;
                pictureBox1.Image = Image.FromFile("images\\h1.jpg");
                
            }
            if (card == 3)
            {
                pictureBox1.Visible = true;
                pictureBox1.Image = Image.FromFile("images\\c1.jpg");
            }
            if (card == 4)
            {
                pictureBox1.Visible = true;
                pictureBox1.Image = Image.FromFile("images\\d1.jpg");
            }
        }

        private void pictureBox1_Click(object sender, EventArgs e)
        {
            if (card != 0) // card = 0 means no card is selected
            {
                if (face == 0) // face = 0 means face-up
                {
                    pictureBox1.Image = Image.FromFile("images\\back.jpg");
                    face = 1; // facing down the card    
                }
                else
                {
                    ShowCard();
                    face = 0; // currently face-up
                }
            }
        }

        public Form1()
        {
            x = 0;
            InitializeComponent();
        }

        private int x;

        private void Form1_Load(object sender, EventArgs e)
        {
            this.DoubleBuffered = true;
            this.Paint += Form1_Paint;
        }

        private void Form1_Paint(object sender, PaintEventArgs e)
        {
            Graphics  g = e.Graphics;
            if (x < this.Size.Width - 1) { x += 1; }
            	else { x = 0; }

            Bitmap bp = new Bitmap(Image.FromFile("images\\h1.jpg"));
            g.DrawImage(bp, x, 25);

            Rectangle Head = new Rectangle(this.Size.Width - 100 - x, 25, 70, 70);
            g.FillEllipse(Brushes.Yellow, Head);

            Pen  b4pen = new Pen(Color.Black, 4);
            g.DrawEllipse(b4pen, Head);
        }
